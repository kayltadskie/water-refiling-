Comprehensive description of all database tables, columns, data types, and constraints.

Table: tb_users

ColumnData TypeConstraintsDescription
userIDINTPK, Auto IncrementUnique identifier for each user
roleENUMNOT NULLUser role: customer, staff, or admin
usernameVARCHAR(50)NOT NULLUsername for login
emailVARCHAR(100)UNIQUE, NOT NULLEmail address for authentication
passwordVARCHAR(255)NOT NULLHashed password
contactVARCHAR(20)NULLContact phone number
created_atTIMESTAMPDEFAULT CURRENT_TIMESTAMPAccount creation date

Table: tb_customers

ColumnData TypeConstraintsDescription
customerIDINTPK, FK → tb_users.userIDLinks to user record
loyaltyPointsINTDEFAULT 0Accumulated reward points
purchaseHistoryTEXTNULLJSON or text of past purchases

Table: tb_administrators

ColumnData TypeConstraintsDescription
adminIDINTPK, FK → tb_users.userIDLinks to user record
adminLevelVARCHAR(20)DEFAULT 'standard'Admin privilege level

Table: tb_products

ColumnData TypeConstraintsDescription
productIDINTPK, Auto IncrementUnique product identifier
productNameVARCHAR(100)NOT NULLName of water product
priceDECIMAL(10,2)NOT NULLPrice per unit
stockQuantityINTDEFAULT 0Available stock count
descriptionTEXTNULLProduct description

Table: tb_orders

ColumnData TypeConstraintsDescription
orderIDINTPK, Auto IncrementUnique order identifier
customerIDINTFK → tb_customers.customerIDCustomer who placed order
orderDateDATENOT NULLDate order was placed
totalAmountDECIMAL(10,2)NOT NULLTotal price of order
orderStatusENUMDEFAULT 'Pending'Pending, Confirmed, Completed, Cancelled
deliveryAddressTEXTNULLDelivery location

ColumnData TypeConstraintsDescription
purchaseIDINTPK, Auto IncrementUnique purchase record ID

productIDINTFK → tb_products.productIDProduct purchased
quantityINTNOT NULLUnits purchased
subtotalDECIMAL(10,2)NOT NULLQuantity × Price

ColumnData TypeConstraintsDescription
rewardIDINTPK, Auto IncrementUnique reward record ID

rewardNameVARCHAR(100)NOT NULLName of reward item
rewardDescriptionTEXTNULLReward details
redeemed_atTIMESTAMPNULLRedemption timestamp

ColumnData TypeConstraintsDescription

ColumnData TypeConstraintsDescription
